ingredients:
  - group: "Japchae"
    items:
      - "6-8oz sweet potato noodles"
      - "2 tsp sesame oil"
      - "1 tbsp canola oil"
      - "1 white onion, diced"
      - "1 red bell pepper, julienned"
      - "2 carrots, diced"
      - "2oz shiitake mushrooms, diced"
      - "3 cloves garlic, minced"
      - "spinach"
  - group: "Chicken"
    items:
      - "2-3 chicken breasts, tenderized & cubed"
      - "2 tbsp soy sauce"
      - "1 tbsp gochujang"
      - "1 tbsp canola oil"
  - group: "Sauce"
    items:
      - "¼ cup soy sauce"
      - "2 tbsp maple syrup"
      - "1 tbsp sesame oil"
      - "Seasonings if desired"
  - group: "Toppings"
    items:
      - "Toasted sesame seeds"
      - "Green onion"
steps:
  - "Mix the soy sauce and gochujang together and then mix in the chicken and let marinate for 30 minutes."
  - "Cook the sweet potato noodles according to their package. Then strain and place in bowl."
  - "Drizzle the sweet potato noodles with sesame oil and mix to ensure the noodles do not stick together."
  - "Cook the chicken in a large pan. Then set aside."
  - "Add the onion to the large pan and cook on medium heat for 3 minutes."
  - "Add in the red bell pepper, carrots, shiitake mushrooms, and garlic. Cook for 5 minutes or until the veggies are soft."
  - "Stir in the spinach and cook an additional 2 minutes."
  - "In a small bowl, mix together the sauce ingredients."
  - "Add the chicken, veggies, and sauce to the bowl of noodles and mix together."
  - "Add toppings if desired."
